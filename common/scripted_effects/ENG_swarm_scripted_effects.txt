ENG_merge_swarm_sisters_pact_effect = {
	if = {
		limit = {
			PRC = { is_in_faction = yes }
		}
		every_other_country = {
			limit = {
				NOT = { tag = ENG }
				OR = {
					tag = PRC
					is_in_faction_with = PRC
					is_subject_of = PRC
				}
			}
			set_country_flag = swarm_merge_eadi_member_flag
		}
	}
	else = {
		PRC = { set_country_flag = swarm_merge_eadi_member_flag }
		every_other_country = {
			limit = { is_subject_of = PRC }
			set_country_flag = swarm_merge_eadi_member_flag
		}
	}

	if = {
		limit = {
			ENG = {
				is_in_faction = yes
				is_faction_leader = yes
			}
		}
		ENG = {
			set_faction_name = swarm_sisters_pact
		}
	}
	else = {
		if = {
			limit = { ENG = { is_in_faction = yes } }
			ENG = { leave_faction = yes }
		}
		ENG = {
			create_faction = swarm_sisters_pact
		}
	}

	PRC = {
		clr_country_flag = swarm_merge_eadi_member_flag
		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}
	}
	ENG = {
		add_to_faction = PRC
	}

	every_other_country = {
		limit = {
			has_country_flag = swarm_merge_eadi_member_flag
		}
		clr_country_flag = swarm_merge_eadi_member_flag
		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}
		ENG = {
			add_to_faction = PREV
		}
	}

	hidden_effect = {
		every_country = {
			limit = {
				is_in_faction_with = ENG
				NOT = { is_in_tech_sharing_group = swarm_sisters_research }
			}
			add_to_tech_sharing_group = swarm_sisters_research
		}
	}
}

ENG_check_swarm_minigame_victory = {
	clamp_variable = { var = ENG_neuro_influence min = 0 max = 100 }
	clamp_variable = { var = ENG_evil_influence min = 0 max = 100 }
	clamp_variable = { var = ENG_vedal_influence min = 0 max = 100 }

	if = {
		limit = {
			has_country_flag = ENG_ogas_minigame_active
			NOT = { has_country_flag = ENG_swarm_minigame_resolved }
		}
		if = {
			limit = {
				check_variable = { var = ENG_neuro_influence value = 100 compare = greater_than_or_equals }
			}
			set_country_flag = ENG_swarm_minigame_resolved
			clr_country_flag = ENG_ogas_minigame_active
			country_event = swarm_uk.10
		}
		else_if = {
			limit = {
				check_variable = { var = ENG_evil_influence value = 100 compare = greater_than_or_equals }
			}
			set_country_flag = ENG_swarm_minigame_resolved
			clr_country_flag = ENG_ogas_minigame_active
			country_event = swarm_uk.11
		}
		else_if = {
			limit = {
				check_variable = { var = ENG_vedal_influence value = 100 compare = greater_than_or_equals }
			}
			set_country_flag = ENG_swarm_minigame_resolved
			clr_country_flag = ENG_ogas_minigame_active
			country_event = swarm_uk.12
		}
	}
}
