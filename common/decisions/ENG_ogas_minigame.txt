	# Timeout 210 days
	ENG_ogas_takeover_mission = {
		icon = generic_ignite_civil_war
		allowed = { original_tag = ENG }
		available = { always = no }
		activation = { always = no }
		cancel_trigger = {
			OR = {
				NOT = { has_country_flag = ENG_ogas_minigame_active }
				is_subject_of = SOV
			}
		}
		days_mission_timeout = 210
		is_good = no

		timeout_effect = {
			if = {
				limit = {
					has_country_flag = ENG_ogas_minigame_active
					NOT = { has_country_flag = ENG_swarm_minigame_resolved }
				}
				set_country_flag = ENG_swarm_minigame_resolved
				clr_country_flag = ENG_ogas_minigame_active
				if = {
					limit = {
						NOT = { check_variable = { ENG_neuro_influence < ENG_evil_influence } }
						NOT = { check_variable = { ENG_neuro_influence < ENG_vedal_influence } }
					}
					country_event = swarm_uk.10
				}
				else_if = {
					limit = {
						check_variable = { ENG_evil_influence > ENG_neuro_influence }
						NOT = { check_variable = { ENG_evil_influence < ENG_vedal_influence } }
					}
					country_event = swarm_uk.11
				}
				else = {
					country_event = swarm_uk.12
				}
			}
		}
	}
