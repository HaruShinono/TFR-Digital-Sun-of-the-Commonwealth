	ENG_project_gundam_mission = {
		icon = generic_army_support
		allowed = { original_tag = ENG }
		available = { always = no }
		activation = { always = no }
		days_mission_timeout = 210
		is_good = yes

		timeout_effect = {
			set_country_flag = ENG_gundam_completed

				limit = { OR = { tag = ENG is_in_faction_with = ENG } }
				set_country_flag = PRC_Gundam
				set_technology = { PRC_Gundam = 1 popup = no }
				add_ideas = ENG_swarm_gundam_corps_idea
			}
			division_template = {
				name = "Swarm Gundam Corps"
				priority = 2
				regiments = {
					PRC_Gundam_DF_41 = { x = 0 y = 0 }
					PRC_Gundam_DF_41 = { x = 0 y = 1 }
					PRC_Gundam_DF_41 = { x = 1 y = 0 }
					PRC_Gundam_DF_41 = { x = 1 y = 1 }
				}
			}
			if = {
				limit = { NOT = { has_dlc = "No Step Back" } }
				add_equipment_production = {
					equipment = { type = PRC_Gundam_equipment_1 creator = "ENG" }
					requested_factories = 5
					progress = 0.5
					amount = 100
				}
			}
			capital_scope = {
				create_unit = {
					division = "name = \"1st Swarm Mobile Suit Wing\" division_template = \"Swarm Gundam Corps\" start_experience_factor = 0.8 start_equipment_factor = 1"
					owner = ENG
					count = 4
				}
			}
		}
	}

	ENG_start_gundam_project_dec = {
		icon = generic_army_support
		allowed = { original_tag = ENG }
		visible = {
			has_country_flag = ENG_unlocked_gundam_rd
			NOT = { has_active_mission = ENG_project_gundam_mission }
			NOT = { has_country_flag = ENG_gundam_completed }
		}
		available = { num_of_military_factories > 4 }
		cost = 35
		complete_effect = {
			add_timed_idea = { idea = ENG_swarm_5_mil_contribution_idea days = 90 }
			activate_mission = ENG_project_gundam_mission
		}
		ai_will_do = { factor = 100 }
	}
