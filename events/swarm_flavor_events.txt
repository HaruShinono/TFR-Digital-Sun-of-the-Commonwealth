# 1. Neuro sends a teasing letter to Vedal while he is skiing in Hokkaido (30 days after taking the throne)
country_event = {
	id = swarm_flavor.1
	title = swarm_flavor.1.t
	desc = swarm_flavor.1.d
	picture = GFX_report_event_generic_snow
	fire_only_once = yes

	trigger = {
		original_tag = ENG
		has_country_flag = ENG_neuro_victory_flag
		JAP = {
			exists = yes
			has_country_flag = JAP_vedal_is_skiing_here
			NOT = { is_subject_of = ENG }
		}
	}

	mean_time_to_happen = { days = 30 }

	option = {
		name = swarm_flavor.1.a # Send a letter to Vedal's phone in Japan!
		add_political_power = 25
		JAP = {
			country_event = { id = swarm_flavor.2 days = 1 }
		}
	}
}
