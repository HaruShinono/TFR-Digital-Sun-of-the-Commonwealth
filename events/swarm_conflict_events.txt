# 1. Conflict of interests: City of London financiers vs. Northern industry
country_event = {
	id = swarm_conflict.1
	title = swarm_conflict.1.t
	desc = swarm_conflict.1.d
	picture = GFX_report_event_generic_rally
	fire_only_once = yes

	trigger = {
		original_tag = ENG
		NOT = { has_country_flag = ENG_neuro_victory_flag }
		OR = {
			has_completed_focus = ENG_tfr_empower_city_london
			has_completed_focus = ENG_tfr_northern_powerhouse_rail
			has_completed_focus = ENG_tfr_mini_budget_meltdown
		}
	}

	mean_time_to_happen = { days = 30 }

	option = {
		name = swarm_conflict.1.a # Concede to London bankers (Northern unions are furious)
		add_political_power = 40
		add_timed_idea = { idea = ENG_conflict_city_vs_north days = 180 }
	}

	option = {
		name = swarm_conflict.1.b # Tax finance to save the North (Markets react harshly)
		add_stability = -0.06
		add_political_power = -50
	}
}

# 3. Random pre-war crisis (If Westminster is in deep decay and stability is below 40%)
country_event = {
	id = swarm_conflict.3
	title = swarm_conflict.3.t
	desc = swarm_conflict.3.d
	picture = GFX_report_event_generic_riot
	fire_only_once = yes

	trigger = {
		original_tag = ENG
		has_stability < 0.40
		has_civil_war = no
		num_owned_states > 4
		OR = {
			has_idea = ENG_tfr_eroding_westminster_2
			has_idea = ENG_tfr_eroding_westminster_3
			has_idea = ENG_tfr_civil_unrest_riots
		}
		NOT = { has_country_flag = ENG_neuro_victory_flag }
	}

	mean_time_to_happen = { days = 40 }

	option = {
		name = swarm_conflict.3.a
		random_list = {
			40 = {
				# General strike and paralyzing riots
				add_stability = -0.08
				add_political_power = -75
			}
			35 = {
				# Disgruntled officers sabotage an arms depot
				add_war_support = -0.10
				add_equipment_to_stockpile = { type = infantry_equipment amount = -2000 }
			}
			25 = {
				# An armed riot escalates into an early localized civil war!
				start_civil_war = {
					ideology = libertarian_socialist
					size = 0.20
					keep_political_leader = yes
					keep_political_party_members = yes
				}
			}
		}
	}
}

# 20. 75% threshold: The anti-AI coalition launches a final counterattack just before Neuro reaches 100%!
country_event = {
	id = swarm_conflict.20
	title = swarm_conflict.20.t
	desc = swarm_conflict.20.d
	picture = GFX_report_event_generic_riot
	fire_only_once = yes

	trigger = {
		original_tag = ENG
		has_country_flag = ENG_ogas_minigame_active
		check_variable = { var = ENG_neuro_influence value = 75 compare = greater_than_or_equals }
	}

	mean_time_to_happen = { days = 2 }

	option = {
		name = swarm_conflict.20.a
		random_list = {
			45 = {
				# Massive DDoS attack on London servers
				add_to_variable = { ENG_neuro_influence = -12 }
				add_to_variable = { ENG_evil_influence = 6 }
				add_to_variable = { ENG_vedal_influence = 6 }
			}
			35 = {
				# Sabotage infrastructure and seize political power
				add_stability = -0.08
				add_political_power = -50
				add_to_variable = { ENG_neuro_influence = -6 }
			}
			20 = {
				# If at peace and stability is low -> A failed military coup turns into fighting!
				if = {
					limit = {
						has_civil_war = no
						has_stability < 0.50
						num_owned_states > 4
					}
					start_civil_war = {
						ideology = conservative
						size = 0.20
						keep_political_leader = yes
						keep_political_party_members = yes
					}
				}
				else = {
					add_stability = -0.10
				}
			}
		}
	}
}
